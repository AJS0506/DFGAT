################################################################################
# DFGAT 모델 학습 명령어
################################################################################

# 사용 가능한 옵션:
# --gpu         : GPU ID (기본값: 1)
# --dataset     : 데이터셋 인덱스 (기본값: 3)
#                 0: movielens_small, 1: movielens_25M, 
#                 2: netflixPrize, 3: filmtrust
# --seed        : 랜덤 시드 (기본값: 1008)
# --emb-dim     : 임베딩 차원 (기본값: 128)
# --first-dim   : 첫번째 레이어 차원 (기본값: 64)
# --second-dim  : 두번째 레이어 차원 (기본값: 32)
# --lr          : 학습률 (기본값: 0.001)
# --early-stop  : Early stopping patience (기본값: 10)
# --epochs      : 최대 에폭 수 (기본값: 200)

# 주의: DFGAT는 Rating 정보가 있는 데이터셋에만 적용 가능
#      (Gowalla는 Rating 없어서 train_gowalla.py 별도 사용)

################################################################################
# 데이터셋 0: MovieLens-Small
################################################################################

# 기본 실행 (모든 기본값 사용)
python3 train_ratingset.py --gpu 0 --dataset 0

# 하이퍼파라미터 튜닝 예시
python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1008 --lr 0.001 --emb-dim 128
python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1008 --lr 0.005 --emb-dim 256
python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1008 --lr 0.01 --emb-dim 64

################################################################################
# 데이터셋 1: MovieLens-25M
################################################################################

# 단일 실행
python3 train_ratingset.py --gpu 1 --dataset 1 --seed 1008

# Early stopping 조정 (대용량 데이터셋)
python3 train_ratingset.py --gpu 1 --dataset 1 --seed 1008 --early-stop 15 --epochs 300

################################################################################
# 데이터셋 2: Netflix Prize
################################################################################

# 단일 실행
python3 train_ratingset.py --gpu 2 --dataset 2 --seed 1008

# 레이어 차원 조정
python3 train_ratingset.py --gpu 2 --dataset 2 --seed 1008 --first-dim 128 --second-dim 64

################################################################################
# 데이터셋 3: FilmTrust
################################################################################

# 단일 실행
python3 train_ratingset.py --gpu 3 --dataset 3 --seed 1008

# 작은 데이터셋용 설정
python3 train_ratingset.py --gpu 3 --dataset 3 --seed 1008 --emb-dim 64 --lr 0.005

################################################################################
# 다중 시드 실험 (재현성 테스트)
################################################################################

# 동일 데이터셋, 다른 시드로 여러 번 실행
for seed in 1008 1009 1010 1011 1012; do
    echo "=============== MovieLens-Small, 시드 $seed 실행 중 ==============="
    python3 train_ratingset.py --gpu 0 --dataset 0 --seed $seed
done

################################################################################
# 다중 GPU 병렬 실행 (각 데이터셋 별도 GPU)
################################################################################

# 4개 데이터셋 동시 실행 (GPU 0~3 사용)
python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1004 &  # MovieLens-Small
python3 train_ratingset.py --gpu 1 --dataset 1 --seed 1004 &  # MovieLens-25M
python3 train_ratingset.py --gpu 2 --dataset 2 --seed 1004 &  # Netflix Prize
python3 train_ratingset.py --gpu 3 --dataset 3 --seed 1004 &  # FilmTrust
wait

################################################################################
# 하이퍼파라미터 그리드 서치
################################################################################

# Learning Rate 실험
for lr in 0.0001 0.0005 0.001 0.005 0.01; do
    echo "=============== LR=$lr 실험 중 ==============="
    python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1008 --lr $lr --epochs 50
done

# Embedding Dimension 실험
for emb in 32 64 128 256; do
    echo "=============== Embedding Dim=$emb 실험 중 ==============="
    python3 train_ratingset.py --gpu 0 --dataset 0 --seed 1008 --emb-dim $emb --epochs 50
done

################################################################################
# 전체 데이터셋 순회 실행
################################################################################

# 모든 데이터셋에 대해 순차 실행
for dataset in 0 1 2 3; do
    seed=$((1008 + dataset))  # 데이터셋별 다른 시드
    echo "=============== 데이터셋 $dataset, 시드 $seed 실행 중 ==============="
    python3 train_ratingset.py --gpu 0 --dataset $dataset --seed $seed
done

################################################################################
# Gowalla 데이터셋 (Rating 없음) - 별도 스크립트 사용
################################################################################

# Gowalla는 train_gowalla.py 사용
python3 train_gowalla.py --gpu 0 --seed 1008

#   - config.json      : 학습 설정
#   - eval.txt         : 평가 결과
#   - *.pth           : 모델 가중치

################################################################################
# 주의사항
################################################################################

# 1. GPU 메모리 부족 시:
#    - --emb-dim, --first-dim, --second-dim 값 줄이기
#    - 배치 크기는 코드 내부에서 자동 조정 (train_set 크기 / 30)

# 2. 학습 시간이 너무 오래 걸릴 때:
#    - --early-stop 값 줄이기 (기본 10)
#    - --epochs 값 줄이기 (기본 200)

# 3. 성능 향상이 필요할 때:
#    - 여러 시드로 실험 후 평균 성능 확인
#    - Learning rate 조정 (0.0001 ~ 0.01 범위)
#    - 임베딩 차원 조정 (32 ~ 256 범위)
